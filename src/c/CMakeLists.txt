cmake_minimum_required(VERSION 3.20)
project(cloudtoid_interprocess VERSION 3.0.2 LANGUAGES C)
include(GNUInstallDirs)
find_program(CARGO_EXECUTABLE cargo REQUIRED)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(CARGO_TARGET_DIR "${CMAKE_CURRENT_BINARY_DIR}/cargo")
if(WIN32)
  set(NATIVE_LIBRARY "${CARGO_TARGET_DIR}/release/cloudtoid_interprocess.dll")
else()
  set(NATIVE_LIBRARY "${CARGO_TARGET_DIR}/release/${CMAKE_SHARED_LIBRARY_PREFIX}cloudtoid_interprocess${CMAKE_SHARED_LIBRARY_SUFFIX}")
endif()
add_custom_target(native ALL COMMAND ${CMAKE_COMMAND} -E env "CARGO_TARGET_DIR=${CARGO_TARGET_DIR}" ${CARGO_EXECUTABLE} build --release --locked -p cloudtoid-interprocess-ffi WORKING_DIRECTORY "${REPO_ROOT}" VERBATIM)
install(FILES "${NATIVE_LIBRARY}" DESTINATION ${CMAKE_INSTALL_LIBDIR})
if(WIN32)
  install(FILES "${NATIVE_LIBRARY}.lib" DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
install(FILES include/interprocess.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
file(RELATIVE_PATH PC_PREFIX "${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_LIBDIR}/pkgconfig" "${CMAKE_INSTALL_PREFIX}")
set(PC_RPATH "")
if(UNIX)
  set(PC_RPATH "-Wl,-rpath,\${libdir}")
endif()
configure_file(cloudtoid-interprocess.pc.in cloudtoid-interprocess.pc @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/cloudtoid-interprocess.pc" DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)

install(FILES "${REPO_ROOT}/LICENSE" DESTINATION ${CMAKE_INSTALL_DATADIR}/cloudtoid-interprocess)
